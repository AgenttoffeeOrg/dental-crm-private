    │                                                                     │
    │  export async function routeDealWithAdapter(context) {             │
    │    const result = await routeDealToPipeline(routingContext)        │
    │                                                                     │
    │    // ===== EMIT EVENT =====                                        │
    │    if (context.contactId) {                                         │
    │      try {                                                          │
    │        await events.dealRouted({                                    │
    │          dealId: context.metadata?.dealId || 'pending',            │
    │          contactId: context.contactId,                             │
    │          tenantId: context.tenantId,                               │
    │          pipelineId: result.pipelineId,                            │
    │          stageId: result.stageId,                                  │
    │          treatmentTags,                                            │
    │          routingMethod: result.routingMethod,                      │
    │          routingLogId: result.routingLogId,                        │
    │          source: context.source                                    │
    │        })                                                           │
    │        console.log('✓ DEAL.ROUTED event emitted')                  │
    │      } catch (error) {                                             │
    │        console.warn('Event emission failed:', error)               │
    │        // Don't fail routing if event fails                        │
    │      }                                                              │
    │    }                                                                │
    │                                                                     │
    │    return result                                                    │
    │  }                                                                  │
    └─────────────────────────────────────────────────────────────────────┘

    ✅ BENEFITS:
    ───────────
    • Automatic: Every routed deal emits event
    • Fault Tolerant: Event failure doesn't break routing
    • Rich Metadata: All routing details included
    • Non-Blocking: Async, no performance impact
